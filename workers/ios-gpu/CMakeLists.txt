cmake_minimum_required(VERSION 3.22)
project(PocketMindMetalPeer LANGUAGES C CXX OBJC OBJCXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_RPC ON CACHE BOOL "" FORCE)
set(GGML_RPC_RDMA OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_DIR}/ggml" ggml)
add_library(PocketMindMetalPeer SHARED entry.cpp PocketMindMetalPeer.h)
target_link_libraries(PocketMindMetalPeer PRIVATE ggml ggml-rpc ggml-metal)
target_link_options(PocketMindMetalPeer PRIVATE "-Wl,-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/exports.txt")
set_target_properties(PocketMindMetalPeer PROPERTIES FRAMEWORK TRUE
  MACOSX_FRAMEWORK_IDENTIFIER org.pocketmind.gpupeer PUBLIC_HEADER PocketMindMetalPeer.h
  XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO INSTALL_NAME_DIR "@rpath")
